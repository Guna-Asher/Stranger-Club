              </div>
            </div>
          </div>
        )}

        {adminView === 'player-details' && selectedPlayer && (
          <div className="p-4">
            <button 
              onClick={() => setAdminView('match-management')}
              className="text-gray-600 hover:text-gray-800 flex items-center space-x-1 mb-4"
            >
              <ArrowLeft className="w-4 h-4" />
              <span>Back</span>
            </button>
            
            <div className="bg-white rounded-lg p-4 shadow-sm">
              <div className="flex items-center justify-between mb-4">
                <h1 className="text-lg font-bold text-gray-900">Player Details</h1>
                <span className={`inline-flex items-center px-2 py-0.5 rounded-full text-xs font-medium ${
                  selectedPlayer.status === 'confirmed' ? 'bg-green-100 text-green-800' :
                  selectedPlayer.status === 'pending' ? 'bg-yellow-100 text-yellow-800' :
                  'bg-red-100 text-red-800'
                }`}>
                  {selectedPlayer.status === 'confirmed' ? 'Confirmed' : 
                   selectedPlayer.status === 'pending' ? 'Payment Pending' : 'Rejected'}
                </span>
              </div>

              <div className="space-y-3 mb-4">
                <div>
                  <label className="block text-xs font-medium text-gray-700 mb-1">Name</label>
                  <p className="text-sm">{selectedPlayer.name}</p>
                </div>
                <div>
                  <label className="block text-xs font-medium text-gray-700 mb-1">Phone</label>
                  <p className="text-sm">{selectedPlayer.phone}</p>
                </div>
                <div>
                  <label className="block text-xs font-medium text-gray-700 mb-1">Email</label>
                  <p className="text-sm">{selectedPlayer.email}</p>
                </div>
                <div>
                  <label className="block text-xs font-medium text-gray-700 mb-1">Amount Paid</label>
                  <p className="text-sm">₹{selectedPlayer.amount}</p>
                </div>
              </div>

              <div className="mb-4">
                <label className="block text-xs font-medium text-gray-700 mb-2">Payment Proof</label>
                <div className="border-2 border-dashed border-gray-300 rounded-lg p-3 text-center">
                  <FileText className="w-8 h-8 mx-auto mb-1 text-gray-400" />
                  <p className="text-xs text-gray-600">Payment screenshot uploaded</p>
                  <p className="text-xs text-gray-500">Submitted at {new Date(selectedPlayer.submittedAt).toLocaleString()}</p>
                </div>
              </div>

              {selectedPlayer.status === 'pending' && (
                <div className="flex space-x-2">
                  <button 
                    onClick={() => handleConfirmPayment(selectedPlayer)}
                    className="flex-1 bg-green-500 hover:bg-green-600 text-white py-2 px-3 rounded text-sm font-medium"
                  >
                    Confirm
                  </button>
                  <button 
                    onClick={() => {
                      setShowRejectModal(true);
                      setSelectedPlayer(player);
                    }}
                    className="flex-1 bg-red-500 hover:bg-red-600 text-white py-2 px-3 rounded text-sm font-medium"
                  >
                    Reject
                  </button>
                </div>
              )}
            </div>
          </div>
        )}
      </div>
    </div>
  );

  // Render current view
  switch (currentView) {
    case 'public-match':
      return <PublicMatchPage />;
    case 'registration':
      return <RegistrationPage />;
    case 'payment':
      return <PaymentPage />;
    case 'payment-submitted':
      return <PaymentSubmittedPage />;
    case 'admin-dashboard':
      return <AdminDashboard />;
    default:
      return <PublicMatchPage />;
  }
};

export default App;
